\begin{abstract}
No straight-line drawing of $K_{14}$ has its edges split into three crossing-free layers, so the geometric
thickness of $K_{14}$ is $4$. Together with the upper bound of Dillencourt, Eppstein and Hirschberg this completes
the values of $\bar\theta(K_n)$ for $n\le 16$. The lower bound is a SAT computation over signotopes with a
hull-size case split and an Euler-type edge bound; two independently written encoders give UNSAT, and every leaf
LRAT proof was checked with \texttt{cake\_lpr}.
\end{abstract}

\section{Introduction}
Dillencourt, Eppstein and Hirschberg \cite{DEH} bounded $\bar\theta(K_n)$ and left $K_{13}$ and $K_{14}$ open.
Seamons showed $\bar\theta(K_{13})=3$ \cite{Seamons}. This note settles $K_{14}$.

\section{Statement}
The geometric thickness $\bar\theta(G)$ is the least $t$ such that $G$ has a straight-line drawing whose edges split
into $t$ crossing-free classes \cite{DEH}.

\begin{theorem}\label{thm:main}
$\bar\theta(K_{14})=4$. Equivalently, no straight-line drawing of $K_{14}$ has its edges split into three
crossing-free layers.
\end{theorem}
\begin{itemize}
\item Upper bound: $\bar\theta(K_n)\le\lceil n/4\rceil$ \cite[Thm.~2.1]{DEH}. An explicit integer drawing with four
  layers: the three-layer $K_{13}$ drawing of \cite{Seamons} plus the point $(1000003,777781)$, whose $13$ edges form
  the fourth layer.
\item Lower bound: new, by the computation below.
\end{itemize}

\begin{corollary}
With \cite{DEH} and \cite{Seamons}, $\bar\theta(K_n)$ is $1$ for $n\le 4$, $2$ for $5\le n\le 8$, $3$ for
$9\le n\le 13$ and $4$ for $14\le n\le 16$.
\end{corollary}

\section{Encoding}
Points $1,\dots,n$ in general position, sorted by $x$; $X(a,b,c)$ is the orientation of $a<b<c$.
\begin{description}
\item[(S)] Signotope axioms on every 4-set \cite{FW,HS}.
\item[(K)] $Y(pq,rs)\leftrightarrow[o(pqr)\ne o(pqs)]\wedge[o(rsp)\ne o(rsq)]$; crossing edges never share a layer.
\item[(N)] Normal form $X(1,a,b)=+$ for $1<a<b$ \cite{HS}; layer value-precedence; (authors' encoder) the reflection
  predicate of \cite{HS}.
\item[(E)] With $h$ hull vertices and $U$ the set of uncrossed edges, $|L_k\cup U|+h\le 3n-3$ for each layer, and
  $(t-1)|U|+th\le t(3n-3)-\binom n2$.
\end{description}

\begin{lemma}\label{lem:sound}
Every straight-line drawing of $K_n$ with $t$ crossing-free layers has an image satisfying (S), (K), (N), (E).
\end{lemma}
\begin{proof}[Proof outline]
\begin{itemize}
\item no collinear triple; rotation to distinct $x$; signotope axioms for $x$-sorted sets;
\item projective map sending a hull vertex first and the rest in angular order; mirror if needed;
\item triangulation edge count $3n-3-h$; uncrossed edges can join any layer; summing over layers.
\end{itemize}
\end{proof}

\section{Case split and certificates}
\begin{itemize}
\item The normal form makes $1$, $2$, $n$ hull vertices. Cases by the other hull vertices $S\subseteq\{3,\dots,13\}$:
  $|S|\le 2$ exactly ($67$ cases) and $|S|\ge3$ by $165$ covers $T\subseteq S$, $|T|=3$; $232$ cases.
\item Optional sweep lemmas for $|S|\le 2$; a second run without them is also UNSAT.
\item Cube-and-conquer with CaDiCaL 3.0.1: $307$ leaves per run. Every leaf LRAT verified by \texttt{cake\_lpr}
  \cite{cake}.
\item Independent encoder and cube partition (referee): $232$ roots, $352$ leaves, all verified.
\item Completeness of each tree is checked separately, and every leaf CNF is regenerated and compared by SHA-256.
\end{itemize}

\section{Data and verification}
Code, case trees with CNF and LRAT hashes, witnesses and a one-command checker: \url{https://github.com/LeeMarshall1113/ai-math-results}, archived on Zenodo, doi:10.5281/zenodo.23748157 (all versions).
\begin{itemize}
\item The LRAT proofs (about $56$\,GB in total) are not archived; \texttt{FULL=1 check\_all.sh} regenerates and
  checks them in about $3.5$ hours on one thread.
\item The encodings' soundness rests on Lemma~\ref{lem:sound} and the case split, proved by hand; there is no formal
  proof that the generators implement them.
\end{itemize}
